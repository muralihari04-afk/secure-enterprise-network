enable
configure terminal

hostname SW1

! =================================
! CREATE VLANs
! =================================

vlan 10
name EMPLOYEES
exit

vlan 20
name SERVERS
exit

vlan 30
name MANAGEMENT
exit

! =================================
! VLAN 10 - EMPLOYEES
! PC1 - PC4
! =================================

interface range fastEthernet 0/1-4
description EMPLOYEE-PCs
switchport mode access
switchport access vlan 10
spanning-tree portfast
no shutdown
exit

! =================================
! VLAN 20 - SERVERS
! Server1 - Server2
! =================================

interface range fastEthernet 0/5-6
description SERVERS
switchport mode access
switchport access vlan 20
no shutdown
exit

! =================================
! VLAN 30 - MANAGEMENT
! PC5 - PC6
! =================================

interface range fastEthernet 0/7-8
description MANAGEMENT-PCs
switchport mode access
switchport access vlan 30
spanning-tree portfast
no shutdown
exit

! =================================
! TRUNK CONNECTION TO R2
! =================================

interface gigabitEthernet 0/1
description TRUNK-TO-R2
switchport mode trunk
no shutdown
exit

! =================================
! MANAGEMENT IP ADDRESS
! =================================

interface vlan 30
ip address 192.168.30.2 255.255.255.0
no shutdown
exit

ip default-gateway 192.168.30.1

! =================================
! SSH CONFIGURATION
! =================================

ip domain-name company.local
username admin privilege 15 secret LabAdmin123!

crypto key generate rsa

! Select 2048 bits when prompted

ip ssh version 2

line vty 0 4
login local
transport input ssh
exit

! =================================
! SAVE CONFIGURATION
! =================================

end
copy running-config startup-config
